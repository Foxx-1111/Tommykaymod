MOD_USA_add_1_effects = {
	if = {
		limit = {
			has_completed_focus = MOD_USA_autonomous_organization_1
		}
		add_ideas = MOD_USA_1_reich_minister_of_armaments
	}
	else_if = {
		limit = {
			has_completed_focus = MOD_USA_ascension_of_number_1
		}
		add_ideas = MOD_USA_1_reich_engineer
	}
}

MOD_USA_remove_1_effects = {
	if = {
		limit = {
			has_idea = MOD_USA_1_reich_minister_of_armaments
		}
		remove_ideas = MOD_USA_1_reich_minister_of_armaments
	}
	else_if = {
		limit = {
			has_idea = MOD_USA_1_reich_engineer
		}
		remove_ideas = MOD_USA_1_reich_engineer
	}
	else = {
		log = "NOTHING TO REMOVE FROM 1"
	}

	if = { #STOP ANY FOCUS IN-PROGRESS (WITH SHINY EFFECT)
		limit = {
			has_shine_effect_on_focus = MOD_USA_form_organization_1
		}
		deactivate_shine_on_focus = MOD_USA_form_organization_1
		clr_country_flag = MOD_USA_form_organization_1_in_progress_flag
	}
	else_if = {
		limit = {
			has_shine_effect_on_focus = MOD_USA_autonomous_organization_1
		}
		deactivate_shine_on_focus = MOD_USA_autonomous_organization_1
		clr_country_flag = MOD_USA_autonomous_organization_1_in_progress_flag
	}
	else_if = {
		limit = {
			has_shine_effect_on_focus = MOD_USA_wunderwaffe
			check_variable = { MOD_USA_wonder_weapons_focus_advisor = 1 }
		}
		deactivate_shine_on_focus = MOD_USA_wunderwaffe
		clr_country_flag = MOD_USA_wunderwaffe_in_progress_flag
		set_variable = { MOD_USA_wonder_weapons_focus_advisor = 0 }
	}

	#Uncomplete Ascension focus so that he can be picked later on
	uncomplete_national_focus = {
		focus = MOD_USA_ascension_of_number_1
		uncomplete_children = no
		refund_political_power = no
	}

	if = { # REMOVE 1 FROM ASCENDED ADVISOR
		limit = {
			check_variable = { MOD_USA_ascended_advisor_1 = 1 }
		}
		set_variable = { MOD_USA_ascended_advisor_1 = 0 }
	}
	else_if = {
		limit = {
			check_variable = { MOD_USA_ascended_advisor_2 = 1 }
		}
		set_variable = { MOD_USA_ascended_advisor_2 = 0 }
	}
	else_if = {
		limit = {
			check_variable = { MOD_USA_ascended_advisor_3 = 1 }
		}
		set_variable = { MOD_USA_ascended_advisor_3 = 0 }
	}
}
